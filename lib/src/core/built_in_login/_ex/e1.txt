
// 1. Task đăng nhập bằng Username & Password
class AppLoginTask extends SimpleLoginTask<LoggedInUserData, EmptyFormInput, CredentialsFormOutput> {
  static const String taskName = "app-login-task";
  final loginRestProvider = LoginProvider();

  AppLoginTask() : super(
    name: taskName,
    formModel: SimpleLoginFormModel<LoggedInUserData>(),
  );

  @override
  Future<ApiResult<LoggedInUserData>> performLogin({
    required CredentialsFormOutput? formData,
  }) async {
    return await loginRestProvider.login(formData!.userName, formData.password);
  }

  @override
  Future<void> onTaskCompleted() async {
    await FlutterArtist.router.off(DashboardScreen.routeName);
  }
}
